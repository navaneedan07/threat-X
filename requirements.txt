# Optional but recommended. As of h5netcdf v1.x the HDF5 backend is an optional
# dependency (it declares only numpy + packaging), so h5netcdf can install and
# still fail with "No module named 'h5py'" -- which is how the precursor pipeline
# silently broke. src/data/loader.py now prefers h5netcdf and falls back to the
# netCDF4 engine (via a temp file) when h5py is absent, so this line is for the
# faster in-memory path rather than a hard requirement.
h5py
zarr
scipy==1.13.0

# GRIB2 reading. The `eccodes` wheel bundles the eccodes binaries, so no

# PyTorch + PyTorch Geometric. These were previously commented out, which left the
# GNN stage unrunnable and made tests/test_gnn_era5_evaluation.py and
# tests/test_gnn_tracking_demo.py fail at collection. They are now pinned: the
# ST-GNN in weights/gnn/05_st_gnn_model.py, the real-ERA5 training in
# weights/gnn/09_train_real_era5.py and the GNN tracking demo all need them.
#
# The CPU wheel is sufficient -- every GNN number in docs/experiments.md was
# produced on CPU in a few minutes, so no GPU is required to reproduce them.
# On Windows/macOS, plain `pip install torch` already resolves to a CPU build.
# For a CUDA build, install torch first from the matching index URL:
#   pip install torch --index-url https://download.pytorch.org/whl/cu124
# and install torch-geometric only AFTER torch, as it has no wheel of its own:
#   pip install torch-geometric
# (torch-geometric is pure Python for the GCNConv layer used here, so no
# compiled extension is needed.)
torch==2.14.0
torch-geometric==2.8.0.post1
